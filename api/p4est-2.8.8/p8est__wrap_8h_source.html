117 /* If \a params.store_adapted is true, these arrays store the indices of the
118 * quadrants refined and coarsened during the most recent call to
119 * \ref p8est_wrap_adapt. The wrap's \a p4est has to be balanced when entering
120 * the adaptation, to avoid multi-level refinement.
121 * The arrays are allocated during the first call of \ref p8est_wrap_adapt.
122 * At every time the arrays index into the local quadrants of the p8est as it
123 * was directly after completion of \ref p8est_wrap_adapt. So, they are not
124 * updated in \ref p8est_wrap_partition. Newly_refined only stores newly
125 * refined quadrants with child id 0. */

437/* Create an iterator over the local leaves in the forest.
438 * Returns a newly allocated state containing the first leaf,
439 * or NULL if the local partition of the tree is empty.
440 * \param [in] pp Legal p8est_wrap structure, hollow or not.
441 * \param [in] track_mirrors If true, \a pp must not be hollow and mirror
442 * information from the ghost layer is stored.
443 * \return NULL if processor is empty, otherwise a leaf iterator for
444 * subsequent use with \a p8est_wrap_leaf_next.
445 */
446p8est_wrap_leaf_t *p8est_wrap_leaf_first (p8est_wrap_t * pp,
447 int track_mirrors);
448
449/* Move the forest leaf iterator forward.
450 * \param [in,out] leaf A non-NULL leaf iterator created by
451 * \ref p8est_wrap_leaf_first.
452 * \return The state that was input with updated information for the
453 * next leaf, or NULL and deallocates the input if called with
454 * the last leaf on this processor.
455 */
