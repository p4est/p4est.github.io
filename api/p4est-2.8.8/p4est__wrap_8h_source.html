458/* Create an iterator over the local leaves in the forest.
459 * Returns a newly allocated state containing the first leaf,
460 * or NULL if the local partition of the tree is empty.
461 * \param [in] pp Legal p4est_wrap structure, hollow or not.
462 * \param [in] track_mirrors If true, \a pp must not be hollow and mirror
463 * information from the ghost layer is stored.
464 * \return NULL if processor is empty, otherwise a leaf iterator for
465 * subsequent use with \a p4est_wrap_leaf_next.
466 */
467p4est_wrap_leaf_t *p4est_wrap_leaf_first (p4est_wrap_t * pp,
468 int track_mirrors);

470/* Move the forest leaf iterator forward.

475 * the last leaf on this processor.
