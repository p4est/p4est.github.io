34/* A p6est_lnodes_t is exactly the same as a p8est_lnodes_t, with the only
35 * difference being that the face_codes are interpreted differently to account
36 * for the types of hanging faces that occur in a p6est. Please see the
37 * documentation for p8est_lnodes */
38
39/* The only other difference is in the numbering of nodes, edges, and faces.
40 *
41 * Columns of nodes are numbered contiguously: this still generates a
42 * partition-unique numbering.
43 *
44 * Although we call a p2est_quadrant_t coordinate layer->z, the orientation of
45 * a layer from lnodes perspective is that the vertical axis is the X axis of
46 * the 3D element, the x axis of the columns is the Y axis of the 3D element,
47 * and the y axis of the columns is the Z axis of the 3D element
48 */
49

115inline int
116p6est_lnodes_decode (p6est_lnodes_code_t face_code, int hanging_face[6],
117 int hanging_edge[12])
118{
119 P4EST_ASSERT (face_code >= 0);
120
121 if (face_code) {
122 /* we pack the p4est_lnodes_code_t at the bottom, followed by a bit
123 * indicating whether this layer is a first or second sibling, followed by
124 * four bits indicating which of the four side faces are layerwise
125 * nonconforming, followed by four bits indicating which of the four side
126 * edges are layerwise nonconforming */
127 p4est_lnodes_code_t fc4 = face_code & 0x000f;
128 int16_t h = (face_code & 0x0010) >> 4;
129 int16_t work = face_code >> 5;
130 int hf;
131 int f, e, w;
132
133 memset (hanging_face, -1, 6 * sizeof (int));
134 memset (hanging_edge, -1, 12 * sizeof (int));
135
136 /* the first two faces are the top and bottom faces, which we know are not
137 * hanging */
138 p4est_lnodes_decode (fc4, hanging_face + 2);
139 for (f = 0; f < 4; f++) {
140 hf = hanging_face[f + 2];
141 w = work & 0x0001;
142 if (hf >= 0) {
143 hanging_edge[p8est_face_edges[f + 2][2]] = 2 + hf;
144 hanging_edge[p8est_face_edges[f + 2][3]] = 2 + hf;
145 hanging_edge[p8est_face_edges[f + 2][1 ^ hf]] = 4;
146 if (w) {
147 hanging_edge[p8est_face_edges[f + 2][3 ^ h]] = 4;
148 hanging_edge[p8est_face_edges[f + 2][1 ^ hf]] = 4;
149 hanging_edge[p8est_face_edges[f + 2][hf]] = 2 + h;
150 hanging_face[f + 2] = (hf << 1) | h;
151 }
152 else {
153 hanging_face[f + 2] = 4 + hf;
154 }
155 }
156 else if (w) {
157 hanging_edge[p8est_face_edges[f + 2][3 ^ h]] = 4;
158 hanging_edge[p8est_face_edges[f + 2][0]] =
159 SC_MAX (hanging_edge[p8est_face_edges[f + 2][0]], 2 + h);
160 hanging_edge[p8est_face_edges[f + 2][1]] =
161 SC_MAX (hanging_edge[p8est_face_edges[f + 2][1]], 2 + h);
162 hanging_face[f + 2] = 6 + h;
163 }
164 work >>= 1;
165 }
166 for (e = 0; e < 4; e++) {
167 if (work & 0x0001) {
168 if (hanging_edge[e] < 0) {
169 hanging_edge[e] = h;
170 }
171#ifdef P4EST_ENABLE_DEBUG
172 else {
173 P4EST_ASSERT (hanging_edge[e] == 2 + h || hanging_edge[e] == 4);
174 }
175#endif
176 }
177 work >>= 1;
178 }
179 return 1;
180 }
181 else {
182 return 0;
183 }
184}
