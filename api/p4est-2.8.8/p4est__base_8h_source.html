384inline unsigned
385p4est_topidx_hash2 (const p4est_topidx_t *tt)
386{
387 uint32_t a, b, c;
388
389#if (P4EST_TOPIDX_FITS_32)
390 a = (uint32_t) tt[0];
391 b = (uint32_t) tt[1];
392 c = 0;
393#else
394 a = (uint32_t) (tt[0] && 0xFFFFFFFF);
395 b = (uint32_t) (tt[0] >> 32);
396 c = (uint32_t) (tt[1] && 0xFFFFFFFF);
397 sc_hash_mix (a, b, c);
398 a += (uint32_t) (tt[1] >> 32);
399#endif
400 sc_hash_final (a, b, c);
401
402 return (unsigned) c;
403}

484inline void
485p4est_topidx_bsort (p4est_topidx_t *t, int length)
486{
487 int i, j;
488 p4est_topidx_t tswap;
489
490 /* go through all elements except the last */
491 for (i = length - 1; i > 0; --i) {
492 /* bubble up the first element until before position i */
493 for (j = 0; j < i; ++j) {
494 if (t[j] > t[j + 1]) {
495 tswap = t[j + 1];
496 t[j + 1] = t[j];
497 t[j] = tswap;
498 }
499 }
500 }
501 P4EST_ASSERT (p4est_topidx_is_sorted (t, length));
502}

522inline uint64_t
523p4est_partition_cut_uint64 (uint64_t global_num, int p, int num_procs)
524{
525 uint64_t result;
526
527 /* In theory, a double * double product should never overflow
528 due to the 15-bit exponent used internally on x87 and above.
529 Also in theory, 80-bit floats should be used internally,
530 and multiply/divide associativity goes left-to-right.
531 Still checking for funny stuff just to be sure. */
532
533 P4EST_ASSERT (0 <= p && p <= num_procs);
534
535 if (p == num_procs) {
536 /* prevent roundoff error and division by zero */
537 return global_num;
538 }
539
540 result = (uint64_t)
541 (((long double) global_num * (double) p) / (double) num_procs);
542
543 P4EST_ASSERT (result <= global_num);
544
545 return result;
546}
