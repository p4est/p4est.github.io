606inline size_t
607sc_array_position (sc_array_t * array, void *element)
608{
609 ptrdiff_t position;
610
611 SC_ASSERT (array->array <= (char *) element);
612 SC_ASSERT (((char *) element - array->array) %
613 (ptrdiff_t) array->elem_size == 0);
614
615 position = ((char *) element - array->array) / (ptrdiff_t) array->elem_size;
616 SC_ASSERT (0 <= position && position < (ptrdiff_t) array->elem_count);
617
618 return (size_t) position;
619}

626inline void *
627sc_array_pop (sc_array_t * array)
628{
629 SC_ASSERT (SC_ARRAY_IS_OWNER (array));
630 SC_ASSERT (array->elem_count > 0);
631
632 return (void *) (array->array + (array->elem_size * --array->elem_count));
633}

639inline void *
640sc_array_push_count (sc_array_t * array, size_t add_count)
641{
642 const size_t old_count = array->elem_count;
643 const size_t new_count = old_count + add_count;
644
645 SC_ASSERT (SC_ARRAY_IS_OWNER (array));
646
647 if (array->elem_size * new_count > (size_t) array->byte_alloc) {
648 sc_array_resize (array, new_count);
649 }
650 else {
651 array->elem_count = new_count;
652 }
653
654 return (void *) (array->array + array->elem_size * old_count);
655}

815inline void *
816sc_mempool_alloc (sc_mempool_t * mempool)
817{
818 void *ret;
819 sc_array_t *freed = &mempool->freed;
820
821 ++mempool->elem_count;
822
823 if (freed->elem_count > 0) {
824 ret = *(void **) sc_array_pop (freed);
825 }
826 else {
827 ret = sc_mstamp_alloc (&mempool->mstamp);
828 if (mempool->zero_and_persist) {
829 memset (ret, 0, mempool->elem_size);
830 }
831 }
832
833#ifdef SC_ENABLE_DEBUG
834 if (!mempool->zero_and_persist) {
835 memset (ret, -1, mempool->elem_size);
836 }
837#endif
838
839 return ret;
840}

846inline void
847sc_mempool_free (sc_mempool_t * mempool, void *elem)
848{
849 sc_array_t *freed = &mempool->freed;
850
851 SC_ASSERT (mempool->elem_count > 0);
852
853#ifdef SC_ENABLE_DEBUG
854 if (!mempool->zero_and_persist) {
855 memset (elem, -1, mempool->elem_size);
856 }
857#endif
858
859 --mempool->elem_count;
860
861 *(void **) sc_array_push (freed) = elem;
862}
